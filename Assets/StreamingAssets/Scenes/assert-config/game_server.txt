async def handle_hall(websocket):
    buf = b""; player_id = None
    try:
        async for msg in websocket:
            buf += msg
            while True:
                packid, body, pkt_len = parse_packet(buf)
                if not packid: break
                buf = buf[2+pkt_len:]

                # 登录验证
                if packid == 201:
                    req = login_pb.LoginReq()
                    req.ParseFromString(body)
                    ok = await internal_verify(req.token, req.player_id)
                    if not ok: await websocket.close(); return
                    player_id = req.player_id
                    try:
                        async with HALL_POOL.acquire() as conn:
                            async with conn.cursor() as cur:
                                await cur.execute("UPDATE player SET last_login_time=%s WHERE player_id=%s", (int(time.time()), player_id))
                    except: pass

                    if player_id not in ONLINE_PLAYERS:
                        ONLINE_PLAYERS.add(player_id)
                        # 保存玩家WS连接
                        if player_id not in PLAYER_WS_MAP:
                            PLAYER_WS_MAP[player_id] = []
                        PLAYER_WS_MAP[player_id].append(websocket)
                        print_with_time(f"玩家进入大厅：{player_id}")
                        if await is_new_player(player_id, HALL_POOL):
                            await init_new_player_item(player_id, HALL_POOL)
                            await send_welcome_email(player_id, HALL_POOL)
                    await send_player_info(websocket, player_id, HALL_POOL)
                    await send_item_list(websocket, player_id, HALL_POOL)
                    await send_email_list(websocket, player_id, HALL_POOL)
                    await send_online_players(websocket)

                # 心跳处理
                elif packid == 203:
                    h_rsp = login_pb.HeartRes()
                    h_rsp.time = int(asyncio.get_event_loop().time())
                    await websocket.send(make_packet(203, h_rsp.SerializeToString()))

                # 获取道具列表 10201
                elif packid == 10201:
                    req = item_pb.ReqLatestItem()
                    req.ParseFromString(body)
                    await websocket.send(await handle_req_latest_item(req.player_id, HALL_POOL))

                # 获取信件列表 10607
                elif packid == 10607:
                    req = email_pb.ReqLatestMail()
                    req.ParseFromString(body)
                    await websocket.send(await handle_req_latest_mail(req.player_id, HALL_POOL))

                # 修改玩家名称 10101
                elif packid == 10101:
                    req = player_pb.ReqChangeNickName()
                    req.ParseFromString(body)
                    await websocket.send(await handle_change_nickname(player_id, req.nickname, HALL_POOL, websocket))

                # 签收全部信件 10601
                elif packid == 10601:
                    req = email_pb.ReadEmailReq()
                    req.ParseFromString(body)
                    await websocket.send(await handle_read_email(player_id, req.guid_list, HALL_POOL, websocket))

                # 进入场景地图 10501
                elif packid == 10501:
                    req = map_hall_pb.ToEnterMapReq()
                    req.ParseFromString(body)
                    await websocket.send(await handle_enter_map(req.player_id, HALL_POOL))

                # 进入对局房间 10701
                elif packid == 10701:
                    req = extral_pb.ToReleaseExtralHallReq()
                    req.ParseFromString(body)
                    await websocket.send(await handle_enter_extral(req.player_id))

                # 对局房间结束回到大厅 10703
                elif packid == 10703:
                    req = extral_pb.ExtralHallBackHallReq()
                    req.ParseFromString(body)
                    await websocket.send(await handle_extral_back_hall(req.player_id))

                # 查看好友 10411
                elif packid == 10411:
                    req = friend_pb.ReqLatestFriendNotic()
                    req.ParseFromString(body)
                    await websocket.send(await handle_req_latest_friend(req.player_id, HALL_POOL))

                # 查看玩家地图 10503
                elif packid == 10503:
                    req = map_hall_pb.GetOtherPlayMapSignReq()
                    req.ParseFromString(body)
                    await websocket.send(await handle_get_other_player_map(req.player_id, HALL_POOL))

                # 【新增】申请匹配游戏 10301
                elif packid == 10301:
                    req = match_pb.MatchGameReq()
                    req.ParseFromString(body)
                    rsp_packet, _ = await handle_match_game_req(req.player_id, req.room_type, HALL_POOL)
                    await websocket.send(rsp_packet)

                # 【新增】取消匹配游戏 10303
                elif packid == 10303:
                    req = match_pb.CancelMatchGameReq()
                    req.ParseFromString(body)
                    await websocket.send(await handle_cancel_match_game_req(req.player_id, req.room_type))

                # 【新增】发送对决邀请 10305
                elif packid == 10305:
                    req = match_pb.AcceptMatchReq()
                    req.ParseFromString(body)
                    rsp_packet, _ = await handle_send_challenge_req(player_id, req.player_id, req.room_type, HALL_POOL)
                    await websocket.send(rsp_packet)

                # 【新增】接受对决邀请 10307
                elif packid == 10307:
                    req = match_pb.AcceptMatchNoticRes()
                    req.ParseFromString(body)
                    await handle_accept_challenge_req(req.player_id, player_id, True, BATTLE_POOL)

                # 【新增】拒绝对决邀请 10308
                elif packid == 10308:
                    req = match_pb.RefuseMatchNoticRes()
                    req.ParseFromString(body)
                    await handle_accept_challenge_req(req.player_id, player_id, False, BATTLE_POOL)

    except Exception as e:
        print_with_time(f"hall异常：{e}")
    finally:
        # 玩家下线清理WS连接
        if player_id in ONLINE_PLAYERS:
            ONLINE_PLAYERS.discard(player_id)
            # 清理WS连接
            if player_id in PLAYER_WS_MAP:
                if websocket in PLAYER_WS_MAP[player_id]:
                    PLAYER_WS_MAP[player_id].remove(websocket)
                if not PLAYER_WS_MAP[player_id]:
                    del PLAYER_WS_MAP[player_id]
            print_with_time(f"玩家退出大厅：{player_id}")
            try:
                async with HALL_POOL.acquire() as conn:
                    async with conn.cursor() as cur:
                        await cur.execute("UPDATE player SET last_logout_time=%s WHERE player_id=%s", (int(time.time()), player_id))
            except: pass

    global HALL_POOL, BATTLE_POOL

                    print_with_time(f"玩家{player_id}进入fight_server，房间ID{room_id}")
                    
                    # 推送房间初始化数据
                    room_id, robot_id, target_player_id, region_id = await send_fight_room_init(websocket, player_id, room_id, FIGHT_POOL)
                    if not room_id:
                        await websocket.close()
                        return
                    
                    # 保存玩家连接
                    PLAYER_WS_MAP[room_id] = websocket
                    
                    # 启动回合调度器
                    if ROBOT_WS and not ROBOT_WS.closed:
                        round_scheduler_task = asyncio.create_task(
                            fight_round_scheduler(room_id, websocket, ROBOT_WS, CARD_TEMPLATE_MAP)
                        )
                
                # 心跳处理
                elif packid == 203:
                    rsp = login_pb.HeartRes()
                    rsp.time = int(time.time())
                    await websocket.send(make_packet(203, rsp.SerializeToString()))
                
                # 退出房间手动结束战斗 50201
                elif packid == 50201:
                    req = record_pb.EndFightRoomBackMapReq()
                    req.ParseFromString(body)
                    await websocket.send(await handle_end_fight_room(req.player_id, req.room_id, req.reason, FIGHT_POOL))
                    # 停止回合调度
                    if round_scheduler_task:
                        round_scheduler_task.cancel()
                    # 清理连接
                    if req.room_id in PLAYER_WS_MAP:
                        del PLAYER_WS_MAP[req.room_id]
                
                # 出牌操作 50101
                elif packid == 50101:
                    req = room_pb.UpPlayCardReq()
                    req.ParseFromString(body)
                    rsp_packet, msg = await handle_up_play_card(req.player_id, req.room_id, req.char_id, req.Card.id, FIGHT_POOL)
                    await websocket.send(rsp_packet)
                
                # 查看剩牌数量 50103
                elif packid == 50103:
                    req = room_pb.GetRemainCardCountReq()
                    req.ParseFromString(body)
                    await websocket.send(await handle_get_remain_card_count(req.player_id, req.room_id))
                
                # 查看手牌信息 50105
                elif packid == 50105:
                    req = room_pb.ReqLatestGamHandCard()
                    req.ParseFromString(body)
                    await websocket.send(await handle_req_latest_gam_hand_card(req.player_id, req.room_id))
    
    except Exception as e:
        print_with_time(f"fight异常：{e}")
    finally:
        # 清理资源
        if round_scheduler_task:
            round_scheduler_task.cancel()
        if room_id and room_id in PLAYER_WS_MAP:
            del PLAYER_WS_MAP[room_id]
        if player_id and room_id:
            await end_fight_room_redis(room_id)
