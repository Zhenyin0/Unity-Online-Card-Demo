## 3. common/network/base_client.py（基础网络+全局变量）
```python
import requests
import json
from struct import pack, unpack
import threading
from common.utils.config_helper import get_client_config

# 加载配置
CONFIG = get_client_config()
BASE_AUTH = CONFIG["BASE_AUTH"]
WS_BIZ = CONFIG["WS_BIZ"]
HEADERS = CONFIG["HEADERS"]

# ===================== 全局变量（原game_test.py全局状态） =====================
LOGIN_TOKEN = None
PLAYER_ID = None
PLAYER_NAME = None
PLAYER_DATA = None
ITEM_DATA = None
WS_STOP_EVENT = threading.Event()

# ===================== 封包规则（原核心封包解包） =====================
PACK_TYPE_WHOLE = 0
MSG_TYPE_CLIENT_REQ = 2

def make_session():
    return 1  # 服务端强制：请求session必须是奇数

def make_packet(pack_id: int, proto_bytes: bytes):
    rpc_header = pack('>BBHI', PACK_TYPE_WHOLE, MSG_TYPE_CLIENT_REQ, pack_id, make_session())
    inner_pkt = rpc_header + proto_bytes
    outer_len = pack('>H', len(inner_pkt))
    return outer_len + inner_pkt

def parse_packet(data: bytes):
    try:
        if len(data) < 2:
            return None, None, None
        pkt_len, = unpack('>H', data[:2])
        if len(data) < 2 + pkt_len:
            return None, None, None
        inner_data = data[2:2+pkt_len]
        if len(inner_data) < 8:
            return None, None, None
        packtype, msgtype, packid, session = unpack('>BBHI', inner_data[:8])
        body = inner_data[8:]
        return packid, body, pkt_len
    except:
        return None, None, None

# ===================== HTTP 注册/登录 =====================
def register(username: str, password: str):
    try:
        url = f"{BASE_AUTH}/user/signup"
        data = {"account": username, "password": password, "channel": 1}
        response = requests.post(url, headers=HEADERS, data=json.dumps(data), timeout=5)
        if response.status_code == 200:
            res = response.json()
            print("✅ 注册成功！" if res.get("code") == 20000 else f"❌ 注册失败：{res.get('message')}")
    except Exception as e:
        print(f"❌ 注册异常：{str(e)}")

def login(username: str, password: str):
    global LOGIN_TOKEN, PLAYER_ID, PLAYER_NAME
    try:
        url = f"{BASE_AUTH}/user/login"
        data = {"account": username, "password": password}
        response = requests.post(url, headers=HEADERS, data=json.dumps(data), timeout=5)
        if response.status_code == 200:
            res = response.json()
            if res.get("code") == 20000:
                print("✅ 登录成功！")
                data = res.get("data", {})
                LOGIN_TOKEN = data.get("token")
                PLAYER_ID = data.get("player_id")
                PLAYER_NAME = data.get("player_name")
                return True
        print(f"❌ 登录失败：{res.get('message', '服务端异常')}")
    except Exception as e:
        print(f"❌ 登录异常：{str(e)}")
    return False
```

## 4. client/handlers/login_handler.py（登录+心跳）
```python
import sys, os
sys.path.insert(0, os.path.abspath(os.path.join(os.path.dirname(__file__), "../..")))

from common.network.base_client import *
from common.utils.logger import write_log
from importlib import import_module
import asyncio

# 加载proto
import_module("common.protos.common.common_pb2")
login_pb = import_module("common.protos.common.2_login_pb2")

async def send_login_req(ws):
    """发送登录请求包"""
    req = login_pb.LoginReq()
    req.token = LOGIN_TOKEN
    req.player_id = PLAYER_ID
    login_pkt = make_packet(201, req.SerializeToString())
    await ws.send(login_pkt)
    write_log("✅ 登录包已发送")

async def send_heartbeat(ws):
    """心跳包循环"""
    heart_pb = import_module("common.protos.common.2_login_pb2")
    while not WS_STOP_EVENT.is_set():
        await asyncio.sleep(5)
        req = heart_pb.HeartReq()
        req.time = int(asyncio.get_event_loop().time())
        pkt = make_packet(203, req.SerializeToString())
        await ws.send(pkt)
```

import sys, os
sys.path.insert(0, os.path.abspath(os.path.join(os.path.dirname(__file__), "../..")))
from common.network.base_client import *
from common.utils.logger import write_log
from importlib import import_module
import os

import_module("common.protos.common.common_pb2")
player_pb = import_module("common.protos.hallserver.101_hallserver_player_pb2")
item_pb = import_module("common.protos.hallserver.102_hallserver_item_pb2")

def handle_player_info(body):
    import common.network.base_client as bc
    global PLAYER_DATA
    info = player_pb.PlayerInfoNotice.FromString(body)
    PLAYER_DATA = {
        "id": bc.PLAYER_ID, "name": bc.PLAYER_NAME,
        "level": info.level, "score": info.rank_score
    }
    write_log(f"✅ 玩家信息 | 等级：{info.level} | 积分：{info.rank_score}")
    if ITEM_DATA:
        save_game_data()

def handle_item_list(body):
    import common.network.base_client as bc
    global ITEM_DATA
    data = item_pb.ItemListNotice.FromString(body)
    ITEM_DATA = [{"id": i.id, "count": i.count} for i in data.item_list]
    write_log(f"✅ 道具信息 | 数量：{len(ITEM_DATA)}")
    if PLAYER_DATA:
        save_game_data()

def save_game_data():
    import common.network.base_client as bc
    try:
        data_filename = f"gameplay-{bc.PLAYER_NAME}.txt"
        with open(data_filename, "w", encoding="utf-8") as f:
            f.write("===== 玩家数据 =====\n")
            f.write(f"ID：{PLAYER_DATA['id']}\n名称：{PLAYER_DATA['name']}\n等级：{PLAYER_DATA['level']}\n积分：{PLAYER_DATA['score']}\n")
            f.write("===== 道具数据 =====\n")
            for item in ITEM_DATA:
                f.write(f"道具ID：{item['id']}  数量：{item['count']}\n")
        write_log("✅ 数据已保存到对应账号文件")
    except Exception as e:
        write_log(f"❌ 保存数据失败：{e}")

from common.utils.logger import write_log
from client.handlers import hall_handler

# 消息包ID映射
MSG_ROUTE = {
    10180: hall_handler.handle_player_info,
    10280: hall_handler.handle_item_list,
}

def dispatch_message(packid, body):
    """分发消息到对应处理器"""

        MSG_ROUTE[packid](body)

## 7. client/network/client_socket.py（WebSocket核心连接）
import asyncio
import websockets
from common.network.base_client import *
from common.utils.logger import write_log
from client.handlers.login_handler import send_login_req, send_heartbeat
from client.network.message_handler import dispatch_message

async def ws_connect():
    """WebSocket主连接逻辑"""
    # 核心修复：实时读取模块原生变量，解决线程不同步
    import common.network.base_client as bc
    if not bc.LOGIN_TOKEN or not bc.PLAYER_ID:
        write_log("❌ 请先登录！")
        return
        
    try:
        async with websockets.connect(
            WS_BIZ,
            ping_interval=None,
            ping_timeout=None,
            compression=None,
            open_timeout=None
        ) as ws:
            write_log("✅ 已连接游戏服务器(11012)")
            asyncio.create_task(send_heartbeat(ws))
            await send_login_req(ws)

            buffer = b""
            while not WS_STOP_EVENT.is_set():
                buffer += await ws.recv()
                while True:
                    packid, body, pkt_len = parse_packet(buffer)
                    if not packid:
                        break
                    buffer = buffer[2 + pkt_len:]
                    dispatch_message(packid, body)

            await ws.close()
            write_log("✅ WebSocket已主动断开")
    except websockets.exceptions.ConnectionClosed:
        write_log("❌ 连接关闭")
    except Exception as e:
        write_log(f"❌ 异常：{e}")

def run_ws_background():
    """后台线程运行WebSocket"""
    WS_STOP_EVENT.clear()
    loop = asyncio.new_event_loop()
    asyncio.set_event_loop(loop)
    try:
        loop.run_until_complete(ws_connect())
    except Exception as e:
        write_log(f"❌ 后台WS异常：{e}")
    finally:
        loop.close()

## 9. client/game_client.py（主入口+菜单）
import sys
import os
import threading
# 添加项目根目录到Python路径（核心：解决导入报错）
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

import asyncio
from common.network.base_client import register, login, WS_STOP_EVENT
import common.network.base_client as bc
from client.network.client_socket import run_ws_background

def gameplay_menu():
    """游戏内菜单"""
    while True:
        print("\n===== 功能菜单 =====")
        print("1. 查看数据")
        print("2. 返回登录界面")
        c = input("请选择：")
        if c == "1":
            data_filename = f"gameplay-{bc.PLAYER_NAME}.txt"
            if os.path.exists(data_filename):
                print(open(data_filename, "r", encoding="utf-8").read())
            else:
                print("❌ 未获取到数据")
        elif c == "2":
            WS_STOP_EVENT.set()
            print("✅ 已断开游戏连接，返回主菜单")
            break

def main():
    """主函数"""
    print("===== 游戏测试工具（账号分文件版） =====")
    while True:
        print("\n1. 注册  2. 登录  3. 退出")
        choice = input("请选择：")
        if choice == "3":
            print("👋 退出")
            break
        info = input("输入 用户名 密码：").strip().split()
        if len(info) != 2:
            continue
        user, pwd = info
        if choice == "1":
            register(user, pwd)
        elif choice == "2":
            if login(user, pwd):
                # 后台线程启动WebSocket
                threading.Thread(target=run_ws_background, daemon=True).start()
                gameplay_menu()

if __name__ == "__main__":
    if sys.platform == 'win32':
        asyncio.set_event_loop_policy(asyncio.WindowsSelectorEventLoopPolicy())
    main()
